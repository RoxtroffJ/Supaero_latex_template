% TITLE PAGE STYLE
% ------------------------------------------------------------------------
\fancypagestyle{title}{
    % Clear header and footer
    \fancyhf{}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}

    % Custom background
        \begin{tikzpicture}[remember picture, overlay]

            % Coordinates of corners with margin
            \coordinate (top left) at ([xshift=\pagetemplatemargin,yshift=-\pagetemplatemargin]current page.north west);
            \coordinate (top right) at ([xshift=-\pagetemplatemargin,yshift=-\pagetemplatemargin]current page.north east);
            \coordinate (bottom left) at ([xshift=\pagetemplatemargin,yshift=\pagetemplatemargin]current page.south west);
            \coordinate (bottom right) at ([xshift=-\pagetemplatemargin,yshift=\pagetemplatemargin]current page.south east);

            % Coordinates of corners with margin and thickness
            \coordinate (top left inner) at ([xshift=\pagetemplatemargin+\pagetemplatethickness,yshift=-\pagetemplatemargin-\pagetemplatethickness]current page.north west);
            \coordinate (top right inner) at ([xshift=-\pagetemplatemargin-\pagetemplatethickness,yshift=-\pagetemplatemargin-\pagetemplatethickness]current page.north east);
            \coordinate (bottom left inner) at ([xshift=\pagetemplatemargin+\pagetemplatethickness,yshift=\pagetemplatemargin+\pagetemplatethickness]current page.south west);
            \coordinate (bottom right inner) at ([xshift=-\pagetemplatemargin-\pagetemplatethickness,yshift=\pagetemplatemargin+\pagetemplatethickness]current page.south east);

            % Draw filled border
            \path [spath/save=outer] 
                (top left) -- (top right) -- (bottom right) -- (bottom left) -- cycle;
            \path [spath/save=inner]
                (top left inner) -- (top right inner) -- (bottom right inner) -- (bottom left inner) -- cycle;
            \fill [
                shade,
                left color=lblue,
                right color=dblue,
                shading angle=-30,
                even odd rule] 
                [spath/use=outer] 
                [spath/use=inner];

            % Logo in the top left corner
            \node[anchor=north west, inner sep=0pt] at (top left) {
                \includegraphics[height=3.8cm]{logos/LOGO_ISAE-SUPAERO_DRAPEAU_GAUCHE_BLEU-NUIT.png}
            };
        
            % Plane drawing
            \pic[scale=62] 
                at ([xshift=83mm,yshift=-65mm]current page.north west) {plane};

            % Hashed area in the bottom right corner
            \pgfmathsetlengthmacro{\hashoffset}{\pagetemplatethickness/tan(\hashangle)}
            \fill[hash, pattern color=dblue]
                ($(bottom left inner)!0.25!(bottom right inner)$) coordinate (hash start) --
                (bottom right inner) --
                ++(0, \pagetemplatethickness) --
                ($(hash start) + (\hashoffset, \pagetemplatethickness)$) --
                cycle;

        \end{tikzpicture}
}


% ------------------------------------------------------------------------
% REDEFINITION OF MAKETITLE
% ------------------------------------------------------------------------

\makeatletter

\renewcommand{\maketitle}{% 
    \vspace*{10mm}    

    \begin{center}

        % Title bold 38pt, tdblue
        {\color{dblue}\bfseries\fontsize{38}{46}\selectfont \@title \par}

        \vspace{5mm}
        % Author 24pt, tlblue
        {\color{lblue}\fontsize{24}{28}\selectfont \@author \par}

        \vspace{5mm}
        % Date 24pt, tlblue
        {\color{lblue}\fontsize{14}{16}\selectfont \@date \par}
    \end{center}

    \vspace{10mm}
}

\makeatother

% ------------------------------------------------------------------------
% MAKE TITLE PAGE command
% ------------------------------------------------------------------------

% Title on a separate, single page. Has custom background and the \maketitle
% in the middle of the page.
% Also has optional content.
% The whole content of the page is centered vertically and horizontally.

\newcommand{\maketitlepage}[1][]{
    \begin{titlepage}
        % Set style for the title page
        \thispagestyle{title}
        
        % Content
        \vspace*{\fill}
        \maketitle
        #1
        \vspace*{\fill}
    \end{titlepage}
}
