\tikzset{
    % Constants
        hashangle/.store in=\hashangle,
        hashangle=50,
        pagetemplatemargin/.store in=\pagetemplatemargin,
        pagetemplatemargin=1cm,
        pagetemplatethickness/.store in=\pagetemplatethickness,
        pagetemplatethickness=0.75cm,
    % Styles
        hash/.style={
            pattern={
                Lines[
                    angle=\hashangle,
                    distance=3pt,
                    line width=0.7pt,
                ]
            }
        },
    % Pics
        pics/plane/.style={
            code = {
                \draw[
                    draw=none,
                    hash,
                    pattern color=lblue,
                    pic actions
                ]
                (0,0) 
                -- (1mm, 0.112903225806452mm) 
                -- (0.112903225806452mm, -0.290322580645161mm) 
                -- (0.0938709677419354mm, -0.102741935483871mm)
                .. controls (0.0841935483870968mm, -0.032258064516129mm)
                .. cycle;
            }
        },
}

% ------------------------------------------------------------------------
% HEADINGS PAGE STYLE
% ------------------------------------------------------------------------
\makeatletter

% Remove capitalization of chapter and section titles in footer
% \renewcommand{\chaptermark}[1]{%
%     \markboth{\chaptername\hspace{0.5em}\thechapter\hspace{0.5em}#1}{}%
% }
\renewcommand{\sectionmark}[1]{%
    \markright{\thesection\hspace{0.5em}#1}%
}
\renewcommand{\subsectionmark}[1]{%
    \markright{\thesubsection\hspace{0.5em}#1}%
}



\fancypagestyle{headings}{
    % Clear header and footer
    \fancyhf{}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}

    % Custom background
    \fancyhead[C]{%
        \begin{tikzpicture}[remember picture, overlay]
    
        % Coordinates of corners with margin
        \coordinate (top left) at ([xshift=\pagetemplatemargin,yshift=-\pagetemplatemargin]current page.north west);
        \coordinate (top right) at ([xshift=-\pagetemplatemargin,yshift=-\pagetemplatemargin]current page.north east);
        \coordinate (bottom left) at ([xshift=\pagetemplatemargin,yshift=\pagetemplatemargin]current page.south west);
        \coordinate (bottom right) at ([xshift=-\pagetemplatemargin,yshift=\pagetemplatemargin]current page.south east);

        % Hashed area in the top left corner
        \pgfmathsetlengthmacro{\hashoffset}{\pagetemplatethickness/tan(\hashangle)}
        \fill[hash, pattern color=lblue]
            (top left) --
            ($(top left)!0.75!(top right)$) coordinate (hash start) --
            ($(hash start) + (-\hashoffset, -\pagetemplatethickness)$) --
            ($(top left) + (0, -\pagetemplatethickness)$) --
            cycle;

        % Logo in the bottom right corner
        \node[anchor=south east, inner sep=0pt] at (bottom right) {
            \includegraphics[height=2.4cm]{logos/LOGO_ISAE-SUPAERO_DRAPEAU_DROITE_DEGRADE.png}
        };

        \end{tikzpicture}%
    }

    % Footer with title, name, chapter (if applicable) section subsection subsubsection, page number
    \fancyfoot[L]{
        % Title
        \color{dblue}\textbf{\@title}

        \vspace{11pt}

        % Chapter number + title if applicable
        % \ifnum\value{chapter}>0
        %     \color{dblue}\leftmark
        % \fi

        \ifnum\value{section}>0
            \color{lblue}\rightmark
        \fi
    }
    \fancyfoot[C]{
        % Author name
        \@author

        % Page number 1cm from the bottom of the page
        \begin{tikzpicture}[remember picture, overlay]
            \node[anchor=south] at ([yshift=\pagetemplatemargin]current page.south) {\thepage};
        \end{tikzpicture}
    }
}

\makeatother
