\makeatletter

\renewcommand{\maketitle}{% 
    \vspace*{10mm}    

    \begin{center}

        % Title bold 38pt, tdblue
        {\color{dblue}\bfseries\fontsize{38}{46}\selectfont \@title \par}

        \vspace{5mm}
        % Author 24pt, tlblue
        {\color{lblue}\fontsize{24}{28}\selectfont \@author \par}

        \vspace{5mm}
        % Date 24pt, tlblue
        {\color{lblue}\fontsize{14}{16}\selectfont \@date \par}
    \end{center}

    \vspace{10mm}
}

\makeatother

% ------------------------------------------------------------------------
% MAKE TITLE PAGE command
% ------------------------------------------------------------------------

% Title on a separate, single page. Has custom background and the \maketitle
% in the middle of the page.
% Also has optional content.
% The whole content of the page is centered vertically and horizontally.

\newcommand{\maketitlepage}[1][]{%
    \thispagestyle{empty} % No header and footer
    \begin{titlepage}
        % Set style for the title page
        \thispagestyle{title}
        
        % Content
        \vspace*{\fill}
        \maketitle
        #1
        \vspace*{\fill}
    \end{titlepage}
}
